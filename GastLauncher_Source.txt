using System.IO;
using System.Windows;
using System.Windows.Threading;

namespace GastLauncher;

public partial class App : Application
{
    private static string LogPath => Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "GastLauncher", "startup.log");

    private void Application_Startup(object sender, StartupEventArgs e)
    {
        DispatcherUnhandledException += (_, args) =>
        {
            WriteError(args.Exception);
            MessageBox.Show("Gast Launcher не смог запуститься. Подробности сохранены в:\n" + LogPath,
                "Ошибка запуска", MessageBoxButton.OK, MessageBoxImage.Error);
            args.Handled = true;
            Shutdown(1);
        };
        AppDomain.CurrentDomain.UnhandledException += (_, args) =>
        {
            if (args.ExceptionObject is Exception ex) WriteError(ex);
        };

        try
        {
            var window = new MainWindow();
            MainWindow = window;
            window.Show();
        }
        catch (Exception ex)
        {
            WriteError(ex);
            MessageBox.Show("Gast Launcher не смог запуститься. Подробности сохранены в:\n" + LogPath,
                "Ошибка запуска", MessageBoxButton.OK, MessageBoxImage.Error);
            Shutdown(1);
        }
    }

    private static void WriteError(Exception ex)
    {
        try
        {
            Directory.CreateDirectory(Path.GetDirectoryName(LogPath)!);
            File.AppendAllText(LogPath, $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] {ex}\n\n");
        }
        catch { }
    }
}

    private void ApplyFilters()
    {

        rows = filter switch
        {
            "Релизы" => rows.Where(v => v.Kind.Equals("release", StringComparison.OrdinalIgnoreCase)),
            "Снапшоты" => rows.Where(v => v.Kind.Contains("snapshot", StringComparison.OrdinalIgnoreCase)),
            "Бета" => rows.Where(v => v.Kind.Contains("beta", StringComparison.OrdinalIgnoreCase)),
            "Альфа" => rows.Where(v => v.Kind.Contains("alpha", StringComparison.OrdinalIgnoreCase)),
            _ => rows
        };

        _visibleVersions.Clear();

        if (_visibleVersions.Count > 0)

    private static void InstallMapArchive(string archivePath, string savesDirectory)
    {
        var temporary = Path.Combine(Path.GetTempPath(), "GastLauncher-map-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(temporary);
        try
        {
            ZipFile.ExtractToDirectory(archivePath, temporary);
            var levelFile = Directory.GetFiles(temporary, "level.dat", SearchOption.AllDirectories).FirstOrDefault();
            if (levelFile is null) throw new InvalidDataException("В архиве карты не найден level.dat.");
            var source = Path.GetDirectoryName(levelFile)!;
            var name = Path.GetFileNameWithoutExtension(archivePath);
            if (string.IsNullOrWhiteSpace(name)) name = "GastLauncher Map";
            var destination = Path.Combine(savesDirectory, SafeFolderToken(name));
            for (var suffix = 2; Directory.Exists(destination); suffix++) destination = Path.Combine(savesDirectory, SafeFolderToken(name) + " (" + suffix + ")");
            CopyDirectory(source, destination);
        }
        finally
        {
            if (Directory.Exists(temporary)) Directory.Delete(temporary, true);
        }
    }

    private static void CopyDirectory(string source, string destination)
    {
        Directory.CreateDirectory(destination);
        foreach (var file in Directory.GetFiles(source)) File.Copy(file, Path.Combine(destination, Path.GetFileName(file)), true);
        foreach (var directory in Directory.GetDirectories(source)) CopyDirectory(directory, Path.Combine(destination, Path.GetFileName(directory)));
    }

    private async Task InstallModrinthPackAsync(ModFileChoice file, string profileDirectory)
    {
        var temporary = Path.Combine(Path.GetTempPath(), "GastLauncher-pack-" + Guid.NewGuid().ToString("N"));
        Directory.CreateDirectory(temporary);
        var archivePath = Path.Combine(temporary, "pack.mrpack");
        try
        {
            await ModApi.DownloadFileAsync(file.DownloadUrl, archivePath);
            using var archive = ZipFile.OpenRead(archivePath);
            var indexEntry = archive.GetEntry("modrinth.index.json") ?? throw new InvalidDataException("Это не корректный Modrinth-модпак: не найден modrinth.index.json.");
            ModrinthPackIndex index;
            await using (var stream = indexEntry.Open())
                index = await JsonSerializer.DeserializeAsync<ModrinthPackIndex>(stream, new JsonSerializerOptions { PropertyNameCaseInsensitive = true })
                    ?? throw new InvalidDataException("Не удалось прочитать список файлов модпака.");

            if (!string.IsNullOrWhiteSpace(index.Dependencies.Minecraft) && index.Dependencies.Minecraft != _activeProfile!.MinecraftVersion)
                throw new InvalidOperationException($"Этот модпак рассчитан на Minecraft {index.Dependencies.Minecraft}, а сборка использует {_activeProfile.MinecraftVersion}. Создай сборку для нужной версии.");
            var expectedLoader = index.Dependencies.FabricLoader is not null ? "fabric" : index.Dependencies.Forge is not null ? "forge" : index.Dependencies.NeoForge is not null ? "neoforge" : index.Dependencies.QuiltLoader is not null ? "quilt" : null;
            if (expectedLoader is not null && !expectedLoader.Equals(_activeProfile!.Loader, StringComparison.OrdinalIgnoreCase))
                throw new InvalidOperationException($"Модпак требует {expectedLoader}, а в выбранной сборке установлен {_activeProfile.Loader}. Создай сборку с нужным загрузчиком.");

            foreach (var packFile in index.Files)
            {
                if (packFile.Env?.Client?.Equals("unsupported", StringComparison.OrdinalIgnoreCase) == true) continue;
                var destination = SafePackPath(profileDirectory, packFile.Path);
                Directory.CreateDirectory(Path.GetDirectoryName(destination)!);
                var downloaded = Path.Combine(temporary, Guid.NewGuid().ToString("N") + ".download");
                try
                {
                    var sourceUrl = packFile.Downloads.FirstOrDefault() ?? throw new InvalidDataException($"Нет ссылки для файла {packFile.Path}.");
                    await ModApi.DownloadFileAsync(sourceUrl, downloaded);
                    if (packFile.Hashes.TryGetValue("sha1", out var expectedSha1))
                    {
                        await using var input = File.OpenRead(downloaded);
                        var actualSha1 = Convert.ToHexString(await System.Security.Cryptography.SHA1.HashDataAsync(input)).ToLowerInvariant();
                        if (!actualSha1.Equals(expectedSha1, StringComparison.OrdinalIgnoreCase)) throw new InvalidDataException($"Проверка файла {packFile.Path} не прошла.");
                    }
                    File.Move(downloaded, destination, true);
                }
                finally { if (File.Exists(downloaded)) File.Delete(downloaded); }
            }

            foreach (var overrideName in new[] { "overrides", "client-overrides" })
            {
                if (!archive.Entries.Any(e => e.FullName.StartsWith(overrideName + "/", StringComparison.Ordinal))) continue;
                var overrideDirectory = Path.Combine(temporary, overrideName);
                Directory.CreateDirectory(overrideDirectory);
                foreach (var entry in archive.Entries.Where(e => e.FullName.StartsWith(overrideName + "/", StringComparison.Ordinal)))
                {
                    var relative = entry.FullName[(overrideName.Length + 1)..];
                    if (string.IsNullOrEmpty(relative)) continue;
                    var target = SafePackPath(overrideDirectory, relative);
                    if (entry.FullName.EndsWith('/')) Directory.CreateDirectory(target);
                    else
                    {
                        Directory.CreateDirectory(Path.GetDirectoryName(target)!);
                        await using var input = entry.Open();
                        await using var output = File.Create(target);
                        await input.CopyToAsync(output);
                    }
                }
                CopyDirectory(overrideDirectory, profileDirectory);
            }
        }
        finally
        {
            if (Directory.Exists(temporary)) Directory.Delete(temporary, true);
        }
    }

    private static string SafePackPath(string root, string relative)
    {
        if (string.IsNullOrWhiteSpace(relative) || Path.IsPathRooted(relative)) throw new InvalidDataException("В архиве найден недопустимый путь.");
        var fullRoot = Path.GetFullPath(root) + Path.DirectorySeparatorChar;
        var fullPath = Path.GetFullPath(Path.Combine(root, relative.Replace('/', Path.DirectorySeparatorChar)));
        if (!fullPath.StartsWith(fullRoot, StringComparison.OrdinalIgnoreCase)) throw new InvalidDataException("В архиве найден путь за пределами папки сборки.");
        return fullPath;
    }

    private async Task InstallModAndRequiredAsync(ModFileChoice file, string modsDirectory, string apiKey, HashSet<string> installed)
    {
        if (!installed.Add(file.ProjectId)) return;
        ModsStatusText.Text = $"Скачиваю {file.DisplayName}…";

        await ModApi.DownloadModAsync(file, apiKey, destination);
        foreach (var dependency in file.RequiredDependencies)
        {
            if (installed.Contains(dependency.ProjectId)) continue;
            ModFileChoice? dependencyFile = null;
            if (file.Platform == "Modrinth")
            {
                if (!string.IsNullOrWhiteSpace(dependency.VersionId))
                    dependencyFile = await ModApi.GetModrinthVersionAsync(dependency.VersionId);
                else
                {
                    var choices = await ModApi.GetModrinthVersionsAsync(dependency.ProjectId, _activeProfile!.MinecraftVersion, _activeProfile.Loader);
                    dependencyFile = choices.FirstOrDefault();
                }
            }
            else
            {
                var choices = await ModApi.GetCurseForgeVersionsAsync(dependency.ProjectId, _activeProfile!.MinecraftVersion, _activeProfile.Loader, apiKey);
                dependencyFile = choices.FirstOrDefault();
            }
            if (dependencyFile is not null)
                await InstallModAndRequiredAsync(dependencyFile, modsDirectory, apiKey, installed);
        }
    }

    private static HttpClient CreateClient(string userAgent)
    {
        var client = new HttpClient { Timeout = TimeSpan.FromSeconds(45) };

        return client;

    public static async Task<List<ModSearchResult>> SearchModrinthAsync(string query, string gameVersion, string loader, string projectType)
    {
        var facetsList = new List<string[]> { new[] { $"project_type:{projectType}" }, new[] { $"versions:{gameVersion}" } };
        if (projectType == "mod") facetsList.Add(new[] { $"categories:{loader.ToLowerInvariant()}" });
        if (projectType == "modpack" && loader.Length > 0) facetsList.Add(new[] { $"categories:{loader.ToLowerInvariant()}" });
        var facets = System.Text.Json.JsonSerializer.Serialize(facetsList);
        var url = $"https://api.modrinth.com/v2/search?query={Uri.EscapeDataString(query)}&limit=30&facets={Uri.EscapeDataString(facets)}";
        using var response = await Modrinth.GetAsync(url);
        response.EnsureSuccessStatusCode();
        var body = await response.Content.ReadFromJsonAsync<ModrinthSearchResponse>() ?? new();
        return body.Hits.Select(x => new ModSearchResult("Modrinth", x.ProjectId, x.Title, x.Description,
            x.Author, x.Downloads, x.Slug, x.IconUrl, projectType)).ToList();
    }

    public static async Task<List<ModFileChoice>> GetModrinthVersionsAsync(string projectId, string gameVersion, string loader, string projectType = "mod")
    {
        var games = Uri.EscapeDataString(System.Text.Json.JsonSerializer.Serialize(new[] { gameVersion }));
        var versionLoader = projectType == "resourcepack" ? "minecraft" : loader.ToLowerInvariant();
        var loaderFilter = projectType is "mod" or "resourcepack" ? $"&loaders={Uri.EscapeDataString(System.Text.Json.JsonSerializer.Serialize(new[] { versionLoader }))}" : "";
        using var response = await Modrinth.GetAsync($"https://api.modrinth.com/v2/project/{Uri.EscapeDataString(projectId)}/version?game_versions={games}{loaderFilter}");
        response.EnsureSuccessStatusCode();
        var versions = await response.Content.ReadFromJsonAsync<List<ModrinthVersion>>() ?? new();
        return versions.Select(v => new ModFileChoice("Modrinth", projectId, v.Id, v.VersionNumber,
            v.Files.FirstOrDefault(f => f.Primary)?.Filename ?? v.Files.FirstOrDefault()?.Filename ?? v.Name,
            v.Name, v.Files.FirstOrDefault(f => f.Primary)?.Url ?? v.Files.FirstOrDefault()?.Url ?? "",
            v.Dependencies.Where(d => projectType == "mod" && d.DependencyType == "required").Select(d => new ModDependency(d.ProjectId, d.VersionId)).ToList(), projectType)).ToList();
    }

    public static async Task<ModFileChoice> GetModrinthVersionAsync(string versionId)
    {
        using var response = await Modrinth.GetAsync($"https://api.modrinth.com/v2/version/{Uri.EscapeDataString(versionId)}");
        response.EnsureSuccessStatusCode();
        var v = await response.Content.ReadFromJsonAsync<ModrinthVersion>() ?? throw new InvalidOperationException("Modrinth не вернул данные версии.");
        var file = v.Files.FirstOrDefault(f => f.Primary) ?? v.Files.FirstOrDefault();
        return new ModFileChoice("Modrinth", v.ProjectId, v.Id, v.VersionNumber, file?.Filename ?? v.Name,
            v.Name, file?.Url ?? "", v.Dependencies.Where(d => d.DependencyType == "required").Select(d => new ModDependency(d.ProjectId, d.VersionId)).ToList());
    }

    public static async Task<List<ModSearchResult>> SearchCurseForgeAsync(string query, string gameVersion, string loader, string apiKey, string projectType)

    public static async Task<List<ModSearchResult>> SearchGitHubModsAsync(string query, string gameVersion, string loader)
    {
        using var client = CreateClient("GastLauncher/1.0 (+https://github.com)");
        client.DefaultRequestHeaders.Accept.ParseAdd("application/vnd.github+json");
        var url = $"https://api.github.com/search/repositories?q={Uri.EscapeDataString(query + " minecraft mod " + gameVersion + " " + loader)}&sort=stars&order=desc&per_page=20";

    public static async Task<List<ModFileChoice>> GetGitHubReleaseAssetsAsync(string repository)
    {
        using var client = CreateClient("GastLauncher/1.0 (+https://github.com)");
        client.DefaultRequestHeaders.Accept.ParseAdd("application/vnd.github+json");
        using var response = await client.GetAsync($"https://api.github.com/repos/{repository}/releases?per_page=10");
        response.EnsureSuccessStatusCode();
        var releases = await response.Content.ReadFromJsonAsync<List<GitHubRelease>>() ?? new();
        return releases.SelectMany(release => release.Assets.Where(asset => asset.Name.EndsWith(".jar", StringComparison.OrdinalIgnoreCase))
            .Select(asset => new ModFileChoice("GitHub Releases", repository, release.TagName, asset.Name, asset.Name,
                $"{release.Name} · {asset.Name}", asset.BrowserDownloadUrl, new(), "mod"))).ToList();
    }

    public static async Task DownloadModAsync(ModFileChoice choice, string apiKey, string destination)
    {
        var url = choice.DownloadUrl;
        if (choice.Platform == "CurseForge")
        {
            var client = CreateCurseClient(apiKey);
            if (string.IsNullOrWhiteSpace(url))
            {
                using var urlResponse = await client.GetAsync($"{CurseForgeBase}/mods/{Uri.EscapeDataString(choice.ProjectId)}/files/{Uri.EscapeDataString(choice.Id)}/download-url");
                urlResponse.EnsureSuccessStatusCode();
                var result = await urlResponse.Content.ReadFromJsonAsync<CurseResponse<string>>();
                url = result?.Data ?? "";
            }
            if (string.IsNullOrWhiteSpace(url)) throw new InvalidOperationException("CurseForge не предоставил прямую ссылку на этот файл.");
            var request = new HttpRequestMessage(HttpMethod.Get, url);
            request.Headers.TryAddWithoutValidation("x-api-key", apiKey);
            using var downloadResponse = await client.SendAsync(request, HttpCompletionOption.ResponseHeadersRead);
            downloadResponse.EnsureSuccessStatusCode();
            await using var input = await downloadResponse.Content.ReadAsStreamAsync();
            await using var output = File.Create(destination);
            await input.CopyToAsync(output);
        }
        else
        {
            if (string.IsNullOrWhiteSpace(url)) throw new InvalidOperationException("У версии мода нет файла для скачивания.");
            using var response = await Modrinth.GetAsync(url, HttpCompletionOption.ResponseHeadersRead);
            response.EnsureSuccessStatusCode();
            await using var input = await response.Content.ReadAsStreamAsync();
            await using var output = File.Create(destination);
            await input.CopyToAsync(output);
        }
    }

    public static async Task DownloadFileAsync(string url, string destination)
    {
        if (!Uri.TryCreate(url, UriKind.Absolute, out var uri) || uri.Scheme != Uri.UriSchemeHttps)
            throw new InvalidOperationException("Ссылка на файл должна быть безопасной HTTPS-ссылкой.");
        using var response = await Modrinth.GetAsync(uri, HttpCompletionOption.ResponseHeadersRead);
        response.EnsureSuccessStatusCode();
        await using var input = await response.Content.ReadAsStreamAsync();
        await using var output = File.Create(destination);
        await input.CopyToAsync(output);
    }

    public static string GetSafeFileName(ModFileChoice choice)

    private void Install_Click(object? sender, EventArgs e)

            using (var resource = OpenResource("GastLauncherSetup.Payload.GastLauncher.exe"))
            using (var output = File.Create(appExe))
                resource.CopyTo(output);
            using (var resource = OpenResource("GastLauncherSetup.Payload.uninstall.ps1"))
            using (var reader = new StreamReader(resource, Encoding.UTF8))
                File.WriteAllText(Path.Combine(installDir, "uninstall.ps1"), reader.ReadToEnd(), Encoding.UTF8);

            var startMenuDir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData), "Microsoft", "Windows", "Start Menu", "Programs", "Gast Launcher");
            Directory.CreateDirectory(startMenuDir);
            CreateShortcut(Path.Combine(startMenuDir, "Gast Launcher.lnk"), appExe, installDir);
            if (_desktopShortcut.Checked)
                CreateShortcut(Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory), "Gast Launcher.lnk"), appExe, installDir);

            using var key = Registry.CurrentUser.CreateSubKey(UninstallKey, true);
            key.SetValue("DisplayName", "Gast Launcher");
            key.SetValue("DisplayVersion", "1.0.0");
            key.SetValue("Publisher", "Gast Launcher");
            key.SetValue("InstallLocation", installDir);
            key.SetValue("DisplayIcon", appExe);
            key.SetValue("UninstallString", $"powershell.exe -NoProfile -ExecutionPolicy Bypass -File \"{Path.Combine(installDir, "uninstall.ps1")}\"");

        $key = [Microsoft.Win32.Registry]::CurrentUser.CreateSubKey($registryPath)
        $key.SetValue('DisplayName', 'Gast Launcher')
        $key.SetValue('DisplayVersion', '1.0.0')
        $key.SetValue('Publisher', 'Gast Launcher')
        $key.SetValue('InstallLocation', $installPath)
        $key.SetValue('UninstallString', ('powershell.exe -NoProfile -ExecutionPolicy Bypass -File "{0}"' -f (Join-Path $installPath 'uninstall.ps1')))
        $key.SetValue('NoModify', 1, [Microsoft.Win32.RegistryValueKind]::DWord)
        $key.SetValue('NoRepair', 1, [Microsoft.Win32.RegistryValueKind]::DWord)
        $key.Close()
        $form.Close()
        if ($launchBox.Checked) { Start-Process (Join-Path $installPath 'GastLauncher.exe') }
    }
    catch {
        $installButton.Enabled = $true
        $installButton.Text = 'Установить'
        [System.Windows.Forms.MessageBox]::Show(('Не удалось установить лаунчер: ' + $_.Exception.Message), 'Gast Launcher', 'OK', 'Error') | Out-Null
    }
})
